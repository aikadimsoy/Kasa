 ```python
import sys
from pathlib import Path
import argparse
import threading
import time
from src.vault.database import Vault
from src.mcp_server.server import app as mcp_app
from src.distill.scheduler import DistillScheduler
from PyQt5.QtWidgets import QApplication
from src.tray.app import KasaTrayApp

def parse_args():
    parser = argparse.ArgumentParser(description="Project KASA")
    parser.add_argument('--vault-path', type=str, default='d:/kasa', help='Path to the vault')
    parser.add_argument('--mcp-port', type=int, default=8000, help='MCP server port')
    parser.add_argument('--no-tray', action='store_true', help='Run in headless mode without tray')
    parser.add_argument('--distill-now', action='store_true', help='Run distillation immediately and exit')
    return parser.parse_args()

def main():
    args = parse_args()
    sys.path.insert(0, str(Path(__file__).parent))

    # Initialize the vault and ensure schema exists.
    vault = Vault(args.vault_path)
    vault.ensure_schema()

    # Start the MCP server in a daemon thread.
    uvicorn_thread = threading.Thread(target=uvicorn.run, args=(mcp_app,), kwargs={"host": "127.0.0.1", "port": args.mcp_port})
    uvicorn_thread.daemon = True
    uvicorn_thread.start()

    # Wait for the server to start (simulating a 1-second grace period).
    time.sleep(1)

    # Start the distillation scheduler in a daemon thread.
    scheduler = DistillScheduler(vault)
    scheduler_thread = threading.Thread(target=scheduler.start)
    scheduler_thread.daemon = True
    scheduler_thread.start()

    if args.distill_now:
        result = scheduler.run_now()
        print(result)
        sys.exit(0)

    if args.no_tray:
        # Block the main thread until a signal is received.
        threading.Event().wait()
    else:
        app = QApplication(sys.argv)
        tray_app = KasaTrayApp(vault, scheduler)
        tray_app.setQuitOnLastWindowClosed(False)
        tray_app.show()
        sys.exit(app.exec_())

if __name__ == "__main__":
    main()
```
